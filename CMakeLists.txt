cmake_minimum_required(VERSION 3.20)
project(ajb_demo VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(AJB_ENABLE_GSTREAMER "Build GStreamer media adapters" ON)
option(AJB_ENABLE_TUI "Build FTXUI interface" ON)
option(AJB_BUILD_TESTS "Build deterministic tests" ON)
option(AJB_ENABLE_ASAN "Enable address/undefined sanitizers" OFF)
option(AJB_ENABLE_TSAN "Enable thread sanitizer" OFF)
if(AJB_ENABLE_ASAN AND AJB_ENABLE_TSAN)
  message(FATAL_ERROR "Use separate ASan and TSan builds")
endif()
if(NOT AJB_ENABLE_GSTREAMER)
  set(AJB_ENABLE_TUI OFF)
endif()
find_package(Threads REQUIRED)
add_library(ajb_core src/core.cpp src/replay.cpp)
target_include_directories(ajb_core PUBLIC include)
target_link_libraries(ajb_core PUBLIC Threads::Threads)
target_compile_options(ajb_core PRIVATE -Wall -Wextra -Wpedantic)
if(AJB_ENABLE_ASAN OR AJB_ENABLE_TSAN)
  target_compile_definitions(ajb_core PUBLIC
    "_LIBCPP_HARDENING_MODE=_LIBCPP_HARDENING_MODE_FAST" _GLIBCXX_ASSERTIONS)
  target_compile_options(ajb_core PUBLIC -fno-omit-frame-pointer -g)
endif()
if(AJB_ENABLE_ASAN)
  target_compile_options(ajb_core PUBLIC -fsanitize=address,undefined)
  target_link_options(ajb_core PUBLIC -fsanitize=address,undefined)
endif()
if(AJB_ENABLE_TSAN)
  target_compile_options(ajb_core PUBLIC -fsanitize=thread)
  target_link_options(ajb_core PUBLIC -fsanitize=thread)
endif()
if(AJB_ENABLE_GSTREAMER)
  if(APPLE)
    execute_process(COMMAND brew --prefix OUTPUT_VARIABLE AJB_BREW_PREFIX
      OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(AJB_BREW_PREFIX)
      set(ENV{PKG_CONFIG_PATH}
        "${AJB_BREW_PREFIX}/lib/pkgconfig:$ENV{PKG_CONFIG_PATH}")
    endif()
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.20
    gstreamer-app-1.0>=1.20 gstreamer-audio-1.0>=1.20)
  set(AJB_GST_PREFIX "")
  foreach(AJB_GST_MODULE gstreamer-1.0 gstreamer-app-1.0 gstreamer-audio-1.0)
    pkg_get_variable(AJB_MODULE_PREFIX ${AJB_GST_MODULE} prefix)
    if(AJB_GST_PREFIX AND NOT AJB_GST_PREFIX STREQUAL AJB_MODULE_PREFIX)
      message(FATAL_ERROR "GStreamer core/app/audio must come from one SDK; fix PKG_CONFIG_PATH")
    endif()
    set(AJB_GST_PREFIX "${AJB_MODULE_PREFIX}")
    if(NOT "${GST_${AJB_GST_MODULE}_VERSION}" STREQUAL "${GST_gstreamer-1.0_VERSION}")
      message(FATAL_ERROR "GStreamer core/app/audio versions differ; use one SDK")
    endif()
  endforeach()
  add_library(ajb_media src/media.cpp src/app.cpp)
  target_link_libraries(ajb_media PUBLIC ajb_core PRIVATE PkgConfig::GST)
  target_compile_options(ajb_media PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(ajb_demo src/main.cpp)
  target_link_libraries(ajb_demo PRIVATE ajb_media)
  if(AJB_ENABLE_TUI)
    find_package(ftxui 7.0.3 EXACT CONFIG REQUIRED)
    target_sources(ajb_demo PRIVATE src/tui.cpp)
    target_link_libraries(ajb_demo PRIVATE
      ftxui::component ftxui::dom ftxui::screen)
    target_compile_definitions(ajb_demo PRIVATE AJB_HAVE_TUI=1)
  endif()
endif()
if(AJB_BUILD_TESTS)
  find_package(GTest 1.18.0 EXACT CONFIG REQUIRED)
  enable_testing()
  add_executable(ajb_tests tests/core_test.cpp tests/replay_test.cpp)
  target_link_libraries(ajb_tests PRIVATE GTest::gtest_main)
  if(AJB_ENABLE_GSTREAMER)
    target_sources(ajb_tests PRIVATE tests/media_test.cpp tests/app_test.cpp)
    target_link_libraries(ajb_tests PRIVATE ajb_media)
  else()
    target_link_libraries(ajb_tests PRIVATE ajb_core)
  endif()
  include(GoogleTest)
  gtest_discover_tests(ajb_tests DISCOVERY_TIMEOUT 30)
endif()
